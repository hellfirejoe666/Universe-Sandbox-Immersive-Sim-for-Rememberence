# ── RMC Helpers (real metrics) ──────────────────────────────────────────────────
def decompose_input(data):
    if isinstance(data, dict): return [{k:v} for k,v in data.items()]
    if isinstance(data, (list,tuple)): return [{f"item_{i}":item} for i,item in enumerate(data)]
    return [data]

def estimate_confidence(result, expected_keys=None):
    if not isinstance(result, (dict,list)): return 0.5 if result is not None else 0.0
    score = 1.0
    if isinstance(result, dict):
        if any(v<0 for v in result.values() if isinstance(v,(int,float))): score -= 0.3
        if expected_keys and len(result)<expected_keys: score *= len(result)/expected_keys
    if isinstance(result,dict) and 'State' in result and abs(result.get('State',0))<5: score += 0.1
    return max(0.0, min(1.0, round(score,2)))

def verify_results(sub_results):
    caveats = []
    for r in sub_results:
        res, conf = r.get("result", {}), r.get("confidence", 0.0)
        if conf < 0.4: caveats.append(f"Low conf sub-result: {conf}")
        if isinstance(res,dict):
            if 'HP' in res and res['HP']<=0: caveats.append("Invalid HP ≤ 0")
            bio_keys = {'MNF','SPL','BEU','STR','FND','KNO','UND','WIS','VIT','SEX','DIV','EGO'}
            if any(k not in bio_keys for k in res if isinstance(res[k],(int,float)) and k in bio_keys):
                caveats.append("Unexpected biorhythm key")
    return {"valid": len(caveats)==0, "caveats": caveats}

def synthesize_results(sub_results):
    merged = {}
    for r in sorted(sub_results, key=lambda x: x.get("confidence",0), reverse=True):
        if isinstance(r.get("result",{}), dict): merged.update(r["result"])
    return merged

def identify_weaknesses(verification): return verification.get("caveats", [])

def refine_input(input_data, weaknesses):
    if isinstance(input_data, dict):
        refined = input_data.copy()
        for w in weaknesses:
            if "missing" in w.lower() or "low confidence" in w.lower():
                if 'biorhythms' in str(input_data):
                    refined.update(dict.fromkeys(['MNF','SPL','BEU','STR','FND','KNO','UND','WIS','VIT','SEX','DIV','EGO'], 0))
        return refined
    return input_data

def recursive_meta_cognition(compute_func, input_data, threshold=0.8, depth=0, max_depth=5):
    if depth > max_depth: return {"result":None, "confidence":0.0, "caveats":["Max depth exceeded"]}
    sub_inputs = decompose_input(input_data)
    sub_results = []
    for sub in sub_inputs:
        res = compute_func(sub)
        conf = estimate_confidence(res, len(sub) if isinstance(sub,dict) else None)
        sub_results.append({"result":res, "confidence":conf})
    verification = verify_results(sub_results)
    synthesized = synthesize_results(sub_results)
    overall_conf = sum(r["confidence"] for r in sub_results)/len(sub_results) if sub_results else 0.0
    if overall_conf < threshold:
        weaknesses = identify_weaknesses(verification)
        refined = refine_input(input_data, weaknesses)
        return recursive_meta_cognition(compute_func, refined, threshold, depth+1, max_depth)
    return {"result":synthesized, "confidence":overall_conf, "caveats":verification.get("caveats",[])}
